\usepackage[most]{tcolorbox}

\newtcolorbox{promptbox}[1][]{%
  enhanced,
  colback   = black!3,      % 底色，想更白就调成 black!2
  colframe  = black!80,     % 边框色
  boxrule   = 0.9pt,        % 边框粗细
  arc       = 7pt,          % 圆角半径，参考图大致是这个量级
  left=10pt, right=10pt, top=9pt, bottom=9pt,
  boxsep    = 0pt,
  breakable,                % 允许跨页跨栏断开
  #1
}

  title=#2, coltitle=white, colbacktitle=black!80,
  fonttitle=\bfseries\small,
  #1
}
